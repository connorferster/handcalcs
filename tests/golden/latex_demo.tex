\subsection*{Logical, identity and membership operators}

\noindent This section demonstrates the \texttt{and}, \texttt{or}, \texttt{not}, \texttt{is}, and \texttt{in} operators. The \texttt{and}/\texttt{or} operators render with their operands spaced, while a nested \texttt{or} inside an \texttt{and} is parenthesized to preserve precedence.

\medskip
\noindent$\displaystyle
\begin{aligned}[t]
\mathrm{temp} &= 75 \quad\text{(Measured temperature, deg C)} \\
\mathrm{temp}_{\mathrm{min}} &= 40 \quad\text{(Minimum operating temperature, deg C)} \\
\mathrm{temp}_{\mathrm{max}} &= 120 \quad\text{(Maximum operating temperature, deg C)} \\
\mathrm{materials} &= [\text{``steel''}, \text{``concrete''}, \text{``timber''}] \\
\mathrm{material} &= \text{``steel''} \\
\mathrm{override} &= \mathrm{None} \\
\mathrm{in}_{\mathrm{range}} &= \mathrm{temp}_{\mathrm{min}} < \mathrm{temp} \;\mathrm{and}\; \mathrm{temp} < \mathrm{temp}_{\mathrm{max}} = 40 < 75 \;\mathrm{and}\; 75 < 120 = \mathrm{True} \quad\text{(Within the operating range)} \\
\mathrm{is}_{\mathrm{extreme}} &= \mathrm{temp} < \mathrm{temp}_{\mathrm{min}} \;\mathrm{or}\; \mathrm{temp} > \mathrm{temp}_{\mathrm{max}} = 75 < 40 \;\mathrm{or}\; 75 > 120 = \mathrm{False} \quad\text{(Outside the operating range)} \\
\mathrm{not}_{\mathrm{frozen}} &= \mathrm{not}\;\left(\mathrm{temp} < 0\right) = \mathrm{not}\;\left(75 < 0\right) = \mathrm{True} \quad\text{(Above freezing)} \\
\mathrm{material}_{\mathrm{ok}} &= \mathrm{material} \in \mathrm{materials} = \text{``steel''} \in [\text{``steel''}, \text{``concrete''}, \text{``timber''}] = \mathrm{True} \quad\text{(Material is recognised)} \\
\mathrm{material}_{\mathrm{missing}} &= \mathrm{material} \notin \mathrm{materials} = \text{``steel''} \notin [\text{``steel''}, \text{``concrete''}, \text{``timber''}] = \mathrm{False} \\
\mathrm{no}_{\mathrm{override}} &= \mathrm{override} \;\mathrm{is}\; \mathrm{None} = \mathrm{None} \;\mathrm{is}\; \mathrm{None} = \mathrm{True} \quad\text{(No manual override was set)} \\
\mathrm{has}_{\mathrm{override}} &= \mathrm{override} \;\mathrm{is\ not}\; \mathrm{None} = \mathrm{None} \;\mathrm{is\ not}\; \mathrm{None} = \mathrm{False} \\
\mathrm{combined} &= \mathrm{in}_{\mathrm{range}} \;\mathrm{and}\; \left(\mathrm{is}_{\mathrm{extreme}} \;\mathrm{or}\; \mathrm{not}_{\mathrm{frozen}}\right) = \mathrm{True} \;\mathrm{and}\; \left(\mathrm{False} \;\mathrm{or}\; \mathrm{True}\right) = \mathrm{True} \quad\text{(Precedence: \texttt{or} is parenthesized)}
\end{aligned}$
